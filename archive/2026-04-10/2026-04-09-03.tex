\begin{mistake}{2026-04-09}{03}

\begin{problemcontent}
\( I = \lim_{x \to 0} \frac{x \sin x^2 - 2(1 - \cos x)\sin x}{x^4} = \) \underline{\hspace{2em}}。
\end{problemcontent}

\begin{solutioncontent}
利用带佩亚诺余项的麦克劳林公式对分子展开至满足精度要求的阶数：
\[
\begin{aligned}
x \sin x^2 &= x(x^2 + o(x^2)) = x^3 + o(x^3) \\
2(1 - \cos x)\sin x &= 2\left(\frac{1}{2}x^2 + o(x^3)\right)(x + o(x^2)) = x^3 + o(x^3)
\end{aligned}
\]
代入原式：
\[
I = \lim_{x \to 0} \frac{x^3 - x^3 + o(x^3)}{x^4} = \lim_{x \to 0} \frac{o(x^3)}{x^4} = 0
\]
（注：若题目分母为 \(x^5\)，则需展开至 5 阶：\(x\sin x^2 = x^3 + o(x^5)\)，\(2(1-\cos x)\sin x = 2(\frac{x^2}{2} - \frac{x^4}{24} + o(x^4))(x - \frac{x^3}{6} + o(x^4)) = x^3 - \frac{1}{4}x^5 + o(x^5)\)，此时相减得 \(\frac{1}{4}x^5 + o(x^5)\)，极限为 \(\frac{1}{4}\)）。
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
 \item \textbf{核心公式}：泰勒展开（\(\sin x \sim x - \frac{1}{6}x^3\)，\(\cos x \sim 1 - \frac{1}{2}x^2 + \frac{1}{24}x^4\)）。
 \item \textbf{易错点}：展开阶数不够，导致高阶无穷小丢失；或者余项四则运算不熟练（如 \(x \cdot o(x^2) = o(x^3)\)）。
 \item \textbf{关联知识}：泰勒展开的"上下同阶"原则。相减抵消的项必须展到抵消后出现的第一个非零项。
\end{itemize}}

\end{mistake}
