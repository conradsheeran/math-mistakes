\begin{mistake}{2026-04-09}{12}

\begin{problemcontent}
已知 \(\displaystyle \int f'(x^3)\,\mathrm{d}x = x^3 + C\)（\(C\) 为任意常数），求 \(f(x)\)。
\end{problemcontent}

\begin{solutioncontent}
对等式两边同时关于 \(x\) 求导：
\[ \frac{\mathrm{d}}{\mathrm{d}x} \left( \int f'(x^3)\,\mathrm{d}x \right) = \frac{\mathrm{d}}{\mathrm{d}x} (x^3 + C) \]
得到：
\[ f'(x^3) = 3x^2 \]

令 \(t = x^3\)，则 \(x = t^{\frac{1}{3}}\)。将其代入上式，得 \(f'\) 的表达式：
\[ f'(t) = 3(t^{\frac{1}{3}})^2 = 3t^{\frac{2}{3}} \]
即 \(f'(x) = 3x^{\frac{2}{3}}\)。

对 \(f'(x)\) 求不定积分即可得到 \(f(x)\)：
\[
\begin{aligned}
f(x) &= \int f'(x) \, \mathrm{d}x \\
&= \int 3x^{\frac{2}{3}} \, \mathrm{d}x \\
&= 3 \cdot \frac{3}{5} x^{\frac{5}{3}} + C_1 \\
&= \frac{9}{5}x^{\frac{5}{3}} + C_1
\end{aligned}
\]
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
 \item \textbf{核心公式/定理}：微积分基本定理 \((\int f(x)\mathrm{d}x)' = f(x)\)；变量代换法。
 \item \textbf{易错点}：混淆 \(f'(x^3)\) 与 \([f(x^3)]'\)。前者是先求导再代入，无链式法则；后者是整体求导，等于 \(f'(x^3) \cdot 3x^2\)。
 \item \textbf{关联知识}：抽象函数求解时，利用换元法统一变量是标准步骤；二次积分时勿忘加上新的积分常数。
\end{itemize}}

\end{mistake}
