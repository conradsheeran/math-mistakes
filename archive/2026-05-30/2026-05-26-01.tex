\begin{mistake}{2026-05-26}{01}

\begin{problemcontent}
求不定积分：
\[
\int \frac{1}{1+\cos x} \, dx
\]
\end{problemcontent}

\begin{solutioncontent}
利用三角函数半角公式进行化简：
\[
1 + \cos x = 2\cos^2 \frac{x}{2}
\]
代入积分中得：
\[
\begin{aligned}
\int \frac{1}{1 + \cos x} \, dx 
&= \int \frac{1}{2\cos^2 \frac{x}{2}} \, dx \\
&= \frac{1}{2} \int \sec^2 \frac{x}{2} \, dx \\
&= \frac{1}{2} \int \sec^2 \frac{x}{2} \cdot 2 \, d\left(\frac{x}{2}\right) \\
&= \tan \frac{x}{2} + C
\end{aligned}
\]
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
  \item \textbf{核心公式/定理}：降幂（半角）公式 \(1 + \cos x = 2\cos^2 \frac{x}{2}\)；基本积分公式 \(\int \sec^2 x \, dx = \tan x + C\)。
  \item \textbf{易错点}：凑微分时易漏掉系数，对 \(\sec^2 \frac{x}{2}\) 积分时需补偿系数 \(2\)，与前方的 \(\frac{1}{2}\) 抵消。
  \item \textbf{关联知识}：处理分母含 \(1 \pm \cos x\) 或 \(1 \pm \sin x\) 的积分时，首选三角恒等变换；次选分子分母同乘共轭项或万能代换法。
\end{itemize}}

\end{mistake}
