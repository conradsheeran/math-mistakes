\begin{mistake}{2026-04-09}{08}

\begin{problemcontent}
求极限 \(\displaystyle \lim_{n \to \infty} x_n \)，其中 \(\displaystyle x_n = \left( \sum_{k=1}^{n} \frac{1}{2(1+2+\cdots+k)} \right)^n\)
\end{problemcontent}

\begin{solutioncontent}
先化简括号内的数列求和部分。
由于分母 \(1+2+\cdots+k = \frac{k(k+1)}{2}\)，通项可化简为：
\[ \frac{1}{2 \cdot \frac{k(k+1)}{2}} = \frac{1}{k(k+1)} = \frac{1}{k} - \frac{1}{k+1} \]
利用裂项相消法求和：
\[ \sum_{k=1}^{n} \left( \frac{1}{k} - \frac{1}{k+1} \right) = \left(1-\frac{1}{2}\right) + \left(\frac{1}{2}-\frac{1}{3}\right) + \cdots + \left(\frac{1}{n}-\frac{1}{n+1}\right) = 1 - \frac{1}{n+1} = \frac{n}{n+1} \]
代回原极限式：
\[
\begin{aligned}
\lim_{n \to \infty} x_n 
&= \lim_{n \to \infty} \left( \frac{n}{n+1} \right)^n \\
&= \lim_{n \to \infty} \left( 1 - \frac{1}{n+1} \right)^n \\
&= \lim_{n \to \infty} \left[ \left( 1 + \frac{-1}{n+1} \right)^{-(n+1)} \right]^{\frac{-n}{n+1}} \\
&= e^{-1} = \frac{1}{e}
\end{aligned}
\]
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
 \item \textbf{核心公式/定理}：等差数列求和 \(S_n=\frac{n(n+1)}{2}\)，重要极限 \(\lim\limits_{x \to 0}(1+x)^{\frac{1}{x}}=e\)
 \item \textbf{易错点}：未先化简前\(n\)项和就直接取极限；处理 \(1^{\infty}\) 型极限时凑指数符号或系数出错。
 \item \textbf{关联知识}：常见裂项公式：\(\frac{1}{n(n+k)}=\frac{1}{k}(\frac{1}{n}-\frac{1}{n+k})\)；\(\frac{n}{(n+1)!}=\frac{1}{n!}-\frac{1}{(n+1)!}\)。
\end{itemize}}

\end{mistake}
