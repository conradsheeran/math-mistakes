\begin{mistake}{2026-05-07}{01}

\begin{problemcontent}
当 \(|x|<1\) 时，求极限
\[
\lim_{n\to\infty}(1+x)(1+x^2)(1+x^4)\cdots(1+x^{2^n}).
\]
\end{problemcontent}

\begin{solutioncontent}
设
\[
P_n=(1+x)(1+x^2)(1+x^4)\cdots(1+x^{2^n}).
\]
由于 \(|x|<1\)，有 \(1-x\ne 0\)。两边乘以 \(1-x\)：
\[
\begin{aligned}
(1-x)P_n
&=(1-x)(1+x)(1+x^2)(1+x^4)\cdots(1+x^{2^n})\\
&=(1-x^2)(1+x^2)(1+x^4)\cdots(1+x^{2^n})\\
&=(1-x^4)(1+x^4)\cdots(1+x^{2^n})\\
&=\cdots\\
&=1-x^{2^{n+1}}.
\end{aligned}
\]
故
\[
P_n=\frac{1-x^{2^{n+1}}}{1-x}.
\]
又因为 \(|x|<1\)，所以 \(x^{2^{n+1}}\to 0\)。因此
\[
\lim_{n\to\infty}P_n
=\lim_{n\to\infty}\frac{1-x^{2^{n+1}}}{1-x}
=\frac{1}{1-x}.
\]
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
  \item \textbf{核心公式/定理}：平方差公式 \((1-a)(1+a)=1-a^2\)。指数为 \(1,2,4,\ldots,2^n\) 的乘积可用连锁消去：\(\prod_{k=0}^{n}(1+x^{2^k})=\frac{1-x^{2^{n+1}}}{1-x}\)。
  \item \textbf{易错点}：不能因为每个因子 \(1+x^{2^k}\to 1\)，就直接判断无限乘积趋于 \(1\)。必须先化为有限乘积的显式表达式再取极限。
  \item \textbf{关联知识}：若 \(|x|<1\)，则 \(x^m\to 0\)；本题中 \(m=2^{n+1}\to\infty\)。
\end{itemize}}

\end{mistake}
