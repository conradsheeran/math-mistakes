\begin{mistake}{2026-05-20}{01}

\begin{problemcontent}
展开 $(a+b)^2$。
\end{problemcontent}

\begin{solutioncontent}
根据完全平方公式，$(a+b)^2 = a^2 + 2ab + b^2$。
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
  \item \textbf{核心公式}：完全平方公式 $(a+b)^2 = a^2 + 2ab + b^2$。
  \item \textbf{易错点}：误认为 $(a+b)^2 = a^2 + b^2$，漏掉中间项 $2ab$。
\end{itemize}}

\end{mistake}
