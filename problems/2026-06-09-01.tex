\begin{mistake}{2026-06-09}{01}

\begin{problemcontent}
求下列函数的导数 \(y'\)：
(1) \(y = (1+x^2)^{\sin x}\)
(2) \(y = \sqrt[3]{\frac{(x+1)(x+2)}{x(1+x^2)}}\)
\end{problemcontent}

\begin{solutioncontent}
\textbf{解(1) (幂指函数求导)：}
两边取自然对数：
\[ \ln y = \sin x \cdot \ln(1+x^2) \]
两边对 \(x\) 求导：
\[ \frac{y'}{y} = \cos x \ln(1+x^2) + \sin x \cdot \frac{2x}{1+x^2} \]
将 \(y\) 乘到右边得到：
\[ y' = (1+x^2)^{\sin x} \left[ \cos x \ln(1+x^2) + \frac{2x \sin x}{1+x^2} \right] \]

\textbf{解(2) (复杂分式求导)：}
两边取绝对值的自然对数并展开：
\[ \ln|y| = \frac{1}{3} \Big( \ln|x+1| + \ln|x+2| - \ln|x| - \ln(1+x^2) \Big) \]
两边对 \(x\) 求导：
\[ \frac{y'}{y} = \frac{1}{3} \left( \frac{1}{x+1} + \frac{1}{x+2} - \frac{1}{x} - \frac{2x}{1+x^2} \right) \]
将 \(y\) 乘到右边得到：
\[ y' = \sqrt[3]{\frac{(x+1)(x+2)}{x(1+x^2)}} \cdot \frac{1}{3} \left( \frac{1}{x+1} + \frac{1}{x+2} - \frac{1}{x} - \frac{2x}{1+x^2} \right) \]
\end{solutioncontent}

\mistakeknowledge{
\begin{itemize}
  \item \textbf{核心公式/定理}：对数求导法。遇到 \(u(x)^{v(x)}\) 或多因子连乘/开方时，先两边取对数再求导。复合对数求导公式为 \( (\ln|u(x)|)' = \frac{u'(x)}{u(x)} \)。
  \item \textbf{易错点}：1. 对未知正负的代数式取对数时必须加绝对值保证有意义（如 \(\ln|x+1|\)）；2. 左边隐函数求导得到的是 \(\frac{y'}{y}\)，最后计算结果极易漏乘原函数 \(y\)。
\end{itemize}}

\end{mistake}
